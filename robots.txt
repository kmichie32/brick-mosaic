# The tuning harness is a developer tool, not a page anyone should land on
# from a search result. Both paths: the host serves it at /harness and
# 307-redirects /harness.html there, so blocking only the .html would miss the
# URL that actually resolves.
Disallow: /harness.html
Disallow: /harness

# Cloudflare Pages serves the whole repo root, and does not read .assetsignore
# (that is a Workers feature). Nothing here is secret — the repo is public — but
# none of it is web content, and letting it be indexed puts source files and
# working notes in search results competing with the actual page.
Disallow: /docs/
Disallow: /test/
Disallow: /scripts/
Disallow: /README.md
Disallow: /CLAUDE.md
Disallow: /package.json
Disallow: /.assetsignore
